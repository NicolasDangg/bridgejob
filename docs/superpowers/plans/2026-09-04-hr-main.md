# HR Main Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local-first recruiter workspace at `/hr` for creating jobs and scorecards, simulating bulk resume processing, reviewing evidence-rich candidates, recording decisions, and exporting CSV.

**Architecture:** `HRMain.tsx` owns browser state and composes the existing dashboard primitives. `hrWorkflow.js` is a pure deterministic domain module for validation, candidate generation, sorting, immutable decision updates, workspace normalization, and CSV serialization. React persists only normalized serializable workspace data; file objects and timer progress remain transient.

**Tech Stack:** React 19, TypeScript/JavaScript ESM, React Router 7, Tailwind CSS 4, Framer Motion, Lucide React, existing shadcn-derived UI primitives, Node test runner.

- [ ] **Step 4: Implement the minimal pure workflow module**

Create `src/components/hrWorkflow.js` with these exports and invariants:

```js
export const EMPTY_WORKSPACE = Object.freeze({ version: 1, selectedJobId: null, jobs: [] });
export const DECISIONS = ['undecided', 'advance', 'hold', 'reject'];
export const SORT_KEYS = ['name', 'score', 'evidence', 'gaps', 'decision'];
const ACCEPTED_EXTENSIONS = ['.zip', '.pdf', '.docx'];

export function validateResumeFiles(files) {
  const names = Array.from(files, (file) => file.name);
  const rejected = names.filter((name) => !ACCEPTED_EXTENSIONS.some((extension) => name.toLowerCase().endsWith(extension)));
  return { valid: names.length > 0 && rejected.length === 0, accepted: names.filter((name) => !rejected.includes(name)), rejected };
}

export function normalizeWorkspace(value) {
  if (!value || value.version !== 1 || !Array.isArray(value.jobs)) return { ...EMPTY_WORKSPACE, jobs: [] };
  const jobs = value.jobs.filter((item) => item && typeof item.id === 'string' && typeof item.title === 'string' && Array.isArray(item.scorecard) && Array.isArray(item.candidates));
  const selectedJobId = jobs.some((item) => item.id === value.selectedJobId) ? value.selectedJobId : jobs[0]?.id ?? null;
  return { version: 1, selectedJobId, jobs };
}

function hashString(value) {
  let hash = 2166136261;
  for (let index = 0; index < value.length; index += 1) {
    hash ^= value.charCodeAt(index);
    hash = Math.imul(hash, 16777619);
  }
  return hash >>> 0;
}

function displayName(filename, fallbackIndex) {
  const stem = filename.replace(/\.(zip|pdf|docx)$/i, '').replace(/[-_]+/g, ' ').trim();
  return stem || `Candidate ${fallbackIndex + 1}`;
}

function buildCandidate(job, sourceFile, candidateIndex, batchId) {
  const seed = hashString(`${job.id}:${sourceFile}:${candidateIndex}:${job.scorecard.map((criterion) => `${criterion.id}:${criterion.weight}`).join('|')}`);
  const assessments = job.scorecard.map((criterion, criterionIndex) => {
    const bucket = (seed + criterionIndex * 17) % 10;
    const status = bucket < 6 ? 'met' : bucket < 8 ? 'unclear' : 'missing';
    const evidence = status === 'met'
      ? `Resume demonstrates relevant ${criterion.name.toLowerCase()} experience.`
      : status === 'unclear'
        ? `Resume mentions related work but does not establish depth in ${criterion.name.toLowerCase()}.`
        : `No clear evidence found for ${criterion.name.toLowerCase()}.`;
    return { criterionId: criterion.id, criterionName: criterion.name, importance: criterion.importance, weight: criterion.weight, status, evidence, contribution: status === 'met' ? criterion.weight : status === 'unclear' ? criterion.weight * 0.5 : 0 };
  });
  const possible = job.scorecard.reduce((sum, criterion) => sum + criterion.weight, 0);
  const earned = assessments.reduce((sum, assessment) => sum + assessment.contribution, 0);
  return {
    id: `${batchId}-${candidateIndex}`,
    name: displayName(sourceFile, candidateIndex),
    sourceFile,
    score: possible ? Math.round((earned / possible) * 100) : 0,
    decision: 'undecided',
    createdAt: new Date(Number.parseInt(batchId.replace(/\D/g, '').slice(-12), 10) || 0).toISOString(),
    assessments,
  };
}

export function createCandidateResults(job, files, batchId) {
  return Array.from(files).flatMap((file, fileIndex) => {
    const count = file.name.toLowerCase().endsWith('.zip') ? 3 : 1;
    return Array.from({ length: count }, (_, archiveIndex) => buildCandidate(
      job,
      file.name,
      fileIndex * 10 + archiveIndex,
      batchId,
    )).map((candidate, archiveIndex) => ({
      ...candidate,
      name: count === 1 ? candidate.name : `${displayName(file.name, fileIndex)} · Candidate ${archiveIndex + 1}`,
    }));
  });
}

export function sortCandidates(candidates, sort) {
  const direction = sort.direction === 'asc' ? 1 : -1;
  const count = (candidate, status) => candidate.assessments.filter((assessment) => assessment.status === status).length;
  return [...candidates].sort((left, right) => {
    let comparison = 0;
    if (sort.key === 'score') comparison = left.score - right.score;
    else if (sort.key === 'evidence') comparison = count(left, 'met') - count(right, 'met');
    else if (sort.key === 'gaps') comparison = count(left, 'missing') + count(left, 'unclear') - count(right, 'missing') - count(right, 'unclear');
    else comparison = String(left[sort.key] ?? '').localeCompare(String(right[sort.key] ?? ''));
    return comparison === 0 ? left.name.localeCompare(right.name) : comparison * direction;
  });
}

export function updateCandidateDecision(workspace, jobId, candidateId, decision) {
  if (!DECISIONS.includes(decision)) return workspace;
  return { ...workspace, jobs: workspace.jobs.map((job) => job.id !== jobId ? job : { ...job, candidates: job.candidates.map((candidate) => candidate.id === candidateId ? { ...candidate, decision } : candidate) }) };
}

function csvCell(value) {
  const text = String(value ?? '');
  return /[",\n\r]/.test(text) ? `"${text.replaceAll('"', '""')}"` : text;
}

export function serializeCandidatesCsv(candidates) {
  const header = ['Candidate', 'Source file', 'Score', 'Evidence count', 'Missing criteria', 'Unclear criteria', 'Decision', 'Evidence details'];
  const rows = candidates.map((candidate) => {
    const byStatus = (status) => candidate.assessments.filter((assessment) => assessment.status === status);
    return [candidate.name, candidate.sourceFile, candidate.score, byStatus('met').length, byStatus('missing').map((item) => item.criterionName).join('; '), byStatus('unclear').map((item) => item.criterionName).join('; '), candidate.decision, candidate.assessments.map((item) => `${item.criterionName}: ${item.evidence}`).join(' | ')];
  });
  return [header, ...rows].map((row) => row.map(csvCell).join(',')).join('\n');
}
```

Use a stable `batchId` supplied by React. Do not read file contents or browser globals in this module.
